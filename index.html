Архитектура БД (PostgreSQL / MySQL) Основные таблицы Фирма «1С» SQL -- Справочник номенклатуры (стекло, рамка, герметик, комплектующие) CREATE TABLE items ( id SERIAL PRIMARY KEY, name VARCHAR(255) NOT NULL, type VARCHAR(50) NOT NULL, -- 'glass', 'spacer', 'sealant', 'desiccant', 'corner' unit VARCHAR(20) NOT NULL, -- 'sqm', 'meter', 'kg', 'pcs' thickness NUMERIC(5,2) DEFAULT 0, -- Толщина для стекла/ширина для рамки cost_price NUMERIC(12,2) NOT NULL ); -- Заказы клиентов CREATE TABLE orders ( id SERIAL PRIMARY KEY, client_name VARCHAR(255) NOT NULL, status VARCHAR(50) DEFAULT 'draft', -- 'draft', 'in_production', 'ready', 'shipped' created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, delivery_date DATE ); -- Позиции заказа (Стеклопакеты) CREATE TABLE order_items ( id SERIAL PRIMARY KEY, order_id INT REFERENCES orders(id) ON DELETE CASCADE, width INT NOT NULL, -- в мм height INT NOT NULL, -- в мм quantity INT NOT NULL DEFAULT 1, formula VARCHAR(100) NOT NULL, -- e.g., '4M1-16-4M1' area_sqm NUMERIC(8,4) GENERATED ALWAYS AS ((width * height)::numeric / 1000000) STORED, perimeter_m NUMERIC(8,4) GENERATED ALWAYS AS (((width + height) * 2)::numeric / 1000) STORED ); -- Структура формулы стеклопакета (слои) CREATE TABLE glass_unit_components ( id SERIAL PRIMARY KEY, order_item_id INT REFERENCES order_items(id) ON DELETE CASCADE, layer_order INT NOT NULL, -- 1, 2, 3... item_id INT REFERENCES items(id), -- Ссылка на стекло или рамку component_type VARCHAR(20) NOT NULL -- 'glass' или 'spacer' ); -- Производственные партии (Оптимизация резки) CREATE TABLE production_batches ( id SERIAL PRIMARY KEY, status VARCHAR(50) DEFAULT 'planned', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); -- Идентификация каждого стеклопакета на производстве (для QR-кодов) CREATE TABLE unit_barcodes ( id SERIAL PRIMARY KEY, order_item_id INT REFERENCES order_items(id), batch_id INT REFERENCES production_batches(id), barcode VARCHAR(100) UNIQUE NOT NULL, current_stage VARCHAR(50) DEFAULT 'cutting', -- 'cutting', 'washing', 'assembly', 'sealing', 'done' pyramid_id INT ); SQL -- Настройки производственных модулей и этапов CREATE TABLE production_settings ( id SERIAL PRIMARY KEY, setting_key VARCHAR(100) UNIQUE NOT NULL, -- Ключ настройки setting_name VARCHAR(255) NOT NULL, -- Человекочитаемое название is_enabled BOOLEAN DEFAULT TRUE, -- Включен/Выключен модуль category VARCHAR(50) DEFAULT 'general' -- Категория (cutting, assembly, stock, etc.) ); -- Заполнение базовыми флагами гибридного производства INSERT INTO production_settings (setting_key, setting_name, is_enabled, category) VALUES ('track_manual_cutting', 'Ручная резка (без загрузки на ЧПУ)', TRUE, 'cutting'), ('auto_deduct_sealant', 'Авто-списание герметика (без весового контроля)', TRUE, 'stock'), ('enable_desiccant_filling_stage', 'Отдельный этап засыпки осушителя', FALSE, 'assembly'), ('use_pyramid_tracking', 'Учёт движения и баланса пирамид', TRUE, 'logistics'), ('require_barcode_scan_assembly', 'Обязательное сканирование ШК на сборке', FALSE, 'assembly'), ('enable_gas_filling_stage', 'Этап газонаполнения (Аргон)', FALSE, 'assembly'); 2. Адаптивная технологическая карта (Routing) Если этап отключён в настройках, система исключает его из цепочки статусов стеклопакета, и изделие сразу переходит на следующий рабочий стол. Пример бизнес-логики на PHP (Laravel / Pure PHP) PHP class ProductionWorkflowService { // Динамическое получение списка этапов для конкретного заказа public function getActiveStages(): array { // Базовый обязательный маршрут $stages = ['cutting']; // Проверяем включен ли этап автоматической мойки/сборки if ($this->isFeatureEnabled('require_barcode_scan_assembly')) { $stages[] = 'washing_assembly'; } // Ручной или автоматический этап вторичной герметизации if ($this->isFeatureEnabled('enable_gas_filling_stage')) { $stages[] = 'gas_filling'; } $stages[] = 'sealing'; // Герметизация if ($this->isFeatureEnabled('use_pyramid_tracking')) { $stages[] = 'pyramid_loading'; // Погрузка на пирамиду } $stages[] = 'ready'; // Готово к отгрузке return $stages; } private function isFeatureEnabled(string $key): bool { // В реальном приложении результат кэшируется в Redis/Memcached return \DB::table('production_settings') ->where('setting_key', $key) ->value('is_enabled') ?? false; } } 3. Гибкая логика списания со склада (Авто / Вручную) На цехах с ручным нанесением герметика или ручной засыпкой молекулярного сита точно отслеживать вес на каждый стеклопакет сложнее. Поэтому списание регулируется флагом: Режим «Автоматический (по нормативу)»: При переходе стеклопакета в статус Done система рассчитывает теорию по формуле и автоматически списывает рамку, бутил, герметик и осушитель. Режим «Ручной / Партионный»: Система не списывает сырьё на каждый единичный пакет, а даёт возможность мастеру цеха в конце смены списать единым документом фактически потраченные бочки герметика или коробки рамки (инвентаризация по остаткам).
